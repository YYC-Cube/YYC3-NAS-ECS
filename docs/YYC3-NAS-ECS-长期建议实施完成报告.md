## 🚀 使用方法

### 1. 基于机器学习的检查

```typescript
import { createMLDocQualityChecker } from '@/lib/ml-doc-quality-checker';

const checker = createMLDocQualityChecker();

const predictions = await checker.checkDocument(content, filePath);

for (const prediction of predictions) {
  console.log(`${prediction.category}: ${prediction.issue}`);
  console.log(`严重程度: ${prediction.severity}`);
  console.log(`置信度: ${prediction.confidence}`);
  if (prediction.suggestion) {
    console.log(`建议: ${prediction.suggestion}`);
  }
}
```

---

### 2. 预测性问题检测

```typescript
import { createPredictiveIssueDetector } from '@/lib/predictive-issue-detector';

const detector = createPredictiveIssueDetector();

const result = await detector.predictIssues(content, {
  type: 'API文档',
  section: '概述',
  length: 5000,
  complexity: 75,
  lastUpdated: new Date(),
  author: 'YYC³'
});

console.log(`风险等级: ${result.riskLevel}`);
console.log(`置信度: ${result.confidence}`);

for (const issue of result.predictedIssues) {
  console.log(`${issue.category}: ${issue.description}`);
  console.log(`可能性: ${issue.likelihood}`);
}
```

---

### 3. 自动化问题修复

```typescript
import { createAutoIssueFixer, DEFAULT_FIX_OPTIONS } from '@/lib/auto-issue-fixer';

const fixer = createAutoIssueFixer();

const results = await fixer.fixIssues(content, undefined, {
  ...DEFAULT_FIX_OPTIONS,
  autoApply: false,
  dryRun: true
});

for (const result of results) {
  console.log(`${result.issueType}: ${result.message}`);
  for (const change of result.changes) {
    console.log(`  - ${change.description}`);
  }
}

const fixedContent = results[results.length - 1].fixedContent;
```
